% 论文排版样式（XeLaTeX + ctex）。字体配置由 build_paper_pdf.py 按本机环境注入，
% 三档依次探测（均为"正文宋体 + 标题黑体 + 西文数字 Times"这一组合，只是来源不同）：
%   1. macOS：SimSong / STHeiti / Times New Roman（系统自带，最贴近样例）
%   2. 装了 fonts-noto-cjk 的机器：思源宋体 Noto Serif CJK SC（正文）
%      + 思源黑体 Noto Sans CJK SC（标题与强调）+ TeX Gyre Termes（西文，
%      Times New Roman 的度量兼容克隆）+ XITS Math（Times 谱系的数学字体）。
%      思源系字形完整、有真正的粗体，观感明显优于 Fandol，是本仓库的默认选择。
%   3. 都没有时回退 TeX Live 自带的 Fandol，保证任何装了 texlive 的机器都能编译。
\usepackage[fontset=none]{ctex}
\usepackage{unicode-math}
@FONT_SETUP@
\usepackage{geometry}
\usepackage{indentfirst}
\usepackage{titlesec}
\usepackage{caption}
\usepackage{float}
\usepackage{setspace}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{graphicx}
\usepackage{fancyhdr}
\usepackage{amsmath,amssymb}
\usepackage{etoolbox}
\usepackage{needspace}   % 表题与表体绑定，见 build_paper_pdf.py
\usepackage{fvextra}
\usepackage{seqsplit}

% 代码逐字环境必须覆盖中文注释、希腊字母与常用校验符号。
% 具体字体由构建脚本按本机可用字库注入，缺少宽字符字体时回退 DejaVu Sans Mono。
@CODE_FONT_SETUP@

% 页面：A4，四边留白不小于 2.5 cm
\geometry{a4paper, top=2.5cm, bottom=2.5cm, left=2.5cm, right=2.5cm}

% 图表前缀中文化
\renewcommand{\figurename}{图}
\renewcommand{\tablename}{表}

% 段落：普通字与加粗字均为 15pt，行距适度收紧，首行缩进两字。
% 正文中文使用宋体，英文与数字使用 Times New Roman（字体配置见构建脚本）。
% 行距按数模论文的常规取 1.5 倍：早先为压页数收到 1.0，版面过密、与样例不符；
% 正文压到 23 页后已有余量，恢复到常规行距仍在 30 页上限内。
\setlength{\parindent}{2em}
\setstretch{1.10}
\setlength{\parskip}{0pt}
\AtBeginDocument{\fontsize{16pt}{20pt}\selectfont}

% 正文重点统一使用黑体粗体；若重点中含数学量，同时启用 \boldmath。
% 不在此命令中改变字号，保证普通字与加粗字始终使用同一字号。
\DeclareTextFontCommand{\textbf}{\bfseries\sffamily\boldmath}

% 页脚居中页码，无页眉横线（页眉不得出现任何身份信息）
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\thepage}\renewcommand{\headrulewidth}{0pt}}

% 标题层级。pandoc 以 --top-level-division=section 转换：
%   Markdown #   -> \section      论文主标题 (26pt)
%   Markdown ##  -> \subsection   一级标题 (22pt)
%   Markdown ### -> \subsubsection 二级标题 (20pt)
%   Markdown ####-> \paragraph     三级标题 (18pt)
% 规则：正文16pt，每加一级标题加一号字。
\titleformat{\section}[block]{\centering\fontsize{26pt}{30pt}\selectfont\bfseries\sffamily}{}{0pt}{}
\titlespacing*{\section}{0pt}{8pt}{10pt}
\titleformat{\subsection}[block]{\centering\fontsize{22pt}{26pt}\selectfont\bfseries\sffamily}{}{0pt}{}
\titlespacing*{\subsection}{0pt}{10pt}{6pt}
\titleformat{\subsubsection}[block]{\fontsize{20pt}{24pt}\selectfont\bfseries\sffamily}{}{0pt}{}
\titlespacing*{\subsubsection}{0pt}{7pt}{3pt}
\titleformat{\paragraph}[block]{\fontsize{18pt}{22pt}\selectfont\bfseries\sffamily}{}{0pt}{}
\titlespacing*{\paragraph}{0pt}{5pt}{2pt}

% 图表题注：小一号字（14pt），加粗、居中、无缩进。
\captionsetup{justification=centering, font={bf}, labelfont=bf,
  singlelinecheck=true, skip=5pt}

% 摘要标题：一级标题同级（22pt 黑体加粗）；摘要正文按 16pt 单倍行距排版。
\newcommand{\paperabstractheading}{%
  \par\vspace{0pt}%
  \begin{center}\fontsize{22pt}{26pt}\selectfont\bfseries\sffamily 摘\hspace{1em}要\end{center}%
  \vspace{2pt}\par}

% 公式编号右对齐
\makeatletter
\def\tagform@#1{\maketag@@@{(#1\@@italiccorr)}}
\makeatother

% 附录源程序：正文的 1.5 倍行距不应传导到代码清单——代码本来就按行阅读，
% 拉开行距只会把附录从 63 页吹到 90 页。这里在逐字环境内把行距压回单倍。
\AtBeginEnvironment{Highlighting}{\setstretch{1}}
\AtBeginEnvironment{verbatim}{\setstretch{1}}
\AtBeginEnvironment{Verbatim}{\setstretch{1}}

% 附录源程序：允许在任意位置断行，避免长行溢出版心；字号压到 7.5pt 以控制篇幅
\fvset{breaklines=true, breakanywhere=true, fontsize=\fontsize{7.5pt}{9pt}\selectfont}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{breaklines=true, breakanywhere=true,
  commandchars=\\\{\}, fontsize=\fontsize{7.5pt}{9pt}\selectfont}

% 表格整体采用五号字和三线表风格。pandoc 生成的 longtable 本身使用 booktabs 横线，
% 这里关闭竖线并收紧列间距，使版面接近 A092 的表格密度。
\setlength{\tabcolsep}{5pt}
\renewcommand{\arraystretch}{1.05}
\AtBeginEnvironment{longtable}{\small\setlength{\LTpre}{3pt}\setlength{\LTpost}{3pt}}
\AtBeginEnvironment{tabular}{\small}
